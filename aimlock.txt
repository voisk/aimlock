-- ПЕРЕТАСКИВАНИЕ ПАНЕЛИ
local dragging = false
local dragStart
local startPos

Panel.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		dragging = true
		dragStart = input.Position
		startPos = Panel.Position
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if dragging and (
		input.UserInputType == Enum.UserInputType.MouseMovement
		or input.UserInputType == Enum.UserInputType.Touch
	) then

		local delta = input.Position - dragStart

		Panel.Position = UDim2.new(
			startPos.X.Scale,
			startPos.X.Offset + delta.X,
			startPos.Y.Scale,
			startPos.Y.Offset + delta.Y
		)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then
		dragging = false
	end
end)

-- SLIDER
local sliderDragging = false

local function SetSliderFromX(x)
	local left = SliderBackground.AbsolutePosition.X
	local width = SliderBackground.AbsoluteSize.X

	local percent = math.clamp(
		(x - left) / width,
		0,
		1
	)

	FOV_RADIUS =
		MIN_FOV + (MAX_FOV - MIN_FOV) * percent

	SliderButton.Position =
		UDim2.new(percent, 0, 0.5, 0)

	UpdateFOV()
end

	sliderDragging = true
end)

		or input.UserInputType == Enum.UserInputType.Touch then

		sliderDragging = true

-- ПРОВЕРКА ВИДНОСТИ
local function IsVisible(Character, Head)
	local origin = Camera.CFrame.Position
	local direction = Head.Position - origin

	local params = RaycastParams.new()
	params.FilterType = Enum.RaycastFilterType.Exclude
	params.FilterDescendantsInstances = {
		LocalPlayer.Character
	}

	local result = workspace:Raycast(origin, direction, params)

	if result then
		return result.Instance:IsDescendantOf(Character)
	end

	return false
end

-- ПОИСК ЦЕЛИ
local function GetClosestHead()
	local closestHead = nil
	local shortestDistance = math.huge

	local viewport = Camera.ViewportSize

	-- Такой же центр, как у круга
	local screenCenter = Vector2.new(
		viewport.X / 2,
		viewport.Y / 2 - 35
	)

	for _, Player in ipairs(Players:GetPlayers()) do
		if Player ~= LocalPlayer and Player.Character then

			local Character = Player.Character
			local Head = Character:FindFirstChild("Head")
			local Humanoid =
				Character:FindFirstChildOfClass("Humanoid")

			if Head and Humanoid and Humanoid.Health > 0 then
				local position, onScreen =
					Camera:WorldToViewportPoint(Head.Position)

				if onScreen then
					local screenPosition =
						Vector2.new(position.X, position.Y)

					local distance =
						(screenPosition - screenCenter).Magnitude

					local insideFOV = true

					if FOVEnabled then
						insideFOV = distance <= FOV_RADIUS
					end

					if insideFOV then
						local canAim = true

						if WallCheckEnabled then
							canAim = IsVisible(Character, Head)
						end

						if canAim and distance < shortestDistance then
							shortestDistance = distance
							closestHead = Head
						end
					end
				end
			end
		end
	end

	return closestHead
end

-- ESP
local function AddESP(Player)
	if Player == LocalPlayer then return end
	if not ESPEnabled then return end
	if not Player.Character then return end

	if Highlights[Player] then
		Highlights[Player]:Destroy()
	end

	local Highlight = Instance.new("Highlight")
	Highlight.Name = "PlayerESP"
	Highlight.Adornee = Player.Character
	Highlight.FillTransparency = 0.5
	Highlight.OutlineTransparency = 0
	Highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
	Highlight.Parent = Player.Character

	Highlights[Player] = Highlight
end

local function RemoveESP(Player)
	if Highlights[Player] then
		Highlights[Player]:Destroy()
		Highlights[Player] = nil
	end
end

local function ToggleESP()
	ESPEnabled = not ESPEnabled

	if ESPEnabled then
		ESPButton.Text = "ESP: ON"
		ESPButton.BackgroundColor3 = Color3.fromRGB(40, 150, 60)

		for _, Player in ipairs(Players:GetPlayers()) do
			AddESP(Player)
		end
	else
		ESPButton.Text = "ESP: OFF"
		ESPButton.BackgroundColor3 = Color3.fromRGB(60, 60, 60)

		for _, Player in ipairs(Players:GetPlayers()) do
			RemoveESP(Player)
		end
	end
end

ESPButton.MouseButton1Click:Connect(ToggleESP)

		WallButton.Text = "WallCheck: OFF"

-- FOV ON / OFF
FOVButton.MouseButton1Click:Connect(function()
	FOVEnabled = not FOVEnabled

	if FOVEnabled then
		FOVButton.Text = "FOV: ON"
		FOVButton.BackgroundColor3 = Color3.fromRGB(40, 150, 60)

		FOVCircle.Visible = true
		SliderBackground.Visible = true
		SliderButton.Visible = true
		FOVLabel.Visible = true
	else
		FOVButton.Text = "FOV: OFF"
		FOVButton.BackgroundColor3 = Color3.fromRGB(60, 60, 60)

		FOVCircle.Visible = false
		SliderBackground.Visible = false
		SliderButton.Visible = false
		FOVLabel.Visible = false
	end
end)

-- НОВЫЕ ИГРОКИ И РЕСПАВН
local function SetupPlayer(Player)
	if Player == LocalPlayer then return end

	Player.CharacterAdded:Connect(function()
		task.wait(0.5)

		if ESPEnabled then
			AddESP(Player)
		end
	end)
end

for _, Player in ipairs(Players:GetPlayers()) do
	SetupPlayer(Player)
end

Players.PlayerAdded:Connect(SetupPlayer)

Players.PlayerRemoving:Connect(function(Player)
	RemoveESP(Player)
end)

-- ОСНОВНОЙ ЦИКЛ
RunService.RenderStepped:Connect(function()
	local viewport = Camera.ViewportSize

	-- КРУГ ЧУТЬ ВЫШЕ ЦЕНТРА
	FOVCircle.Position = UDim2.fromOffset(
		viewport.X / 2,
		viewport.Y / 2 - 35
	)

	-- AIMLOCK
	if AimLockEnabled then
		local Head = GetClosestHead()

		if Head then
			Camera.CFrame = CFrame.new(
				Camera.CFrame.Position,
				Head.Position
			)
		end
	end
end)
